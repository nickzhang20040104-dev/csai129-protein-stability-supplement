"""Frozen canonical-reference experiment. See sequence_audit.json for limitations."""
import json, sys, time
from pathlib import Path
import numpy as np
import torch

b = Path('<PROTEIN_TOOLS_ROOT>')
root = Path(__file__).resolve().parent
model_name = sys.argv[1]
torch.set_num_threads(16 if model_name == 'esmfold2' else 8)
inputs = json.loads((root / 'sequences.json').read_text())
seeds = [42, 43, 44]
if model_name == 'esmfold2':
    from esm.models.esmfold2 import EsmFold2Model, ESMFold2InputBuilder, ProteinInput, StructurePredictionInput
    model = EsmFold2Model.from_pretrained(str(b/'weights/esmfold2'), device='cuda', load_esmc=False).eval()
    model.load_esmc(str(b/'weights/esmc-6b'))
    builder = ESMFold2InputBuilder(ccd_cache=b/'weights/esmfold2')
else:
    from chai_lab.chai1 import run_inference
for item in inputs:
    for seed in seeds:
        out = root/model_name/item['sequence_id']/str(seed)
        out.mkdir(parents=True, exist_ok=True)
        if (out/'complete.json').exists():
            continue
        start = time.time()
        if model_name == 'esmfold2':
            inp = StructurePredictionInput(sequences=[ProteinInput(id='A', sequence=item['sequence'])])
            result = builder.fold(model, inp, num_loops=20, num_sampling_steps=200, num_diffusion_samples=1, seed=seed)
            (out/'prediction.cif').write_text(result.complex.to_mmcif())
            raw = result.plddt
        else:
            fasta = out/'input.fasta'
            fasta.write_text('>protein|name=A\n'+item['sequence']+'\n')
            result = run_inference(fasta_file=fasta, output_dir=out/'prediction', num_trunk_recycles=3, num_diffn_timesteps=200, num_diffn_samples=1, seed=seed, device='cuda:0', use_esm_embeddings=True, use_msa_server=False, use_templates_server=False)
            assert len(result.cif_paths) == 1
            raw = result.plddt
        if hasattr(raw, 'detach'):
            raw = raw.detach().cpu().numpy()
        arr = np.asarray(raw)
        np.save(out/'plddt_native.npy', arr)
        assert np.isfinite(arr).all()
        assert arr.size == len(item['sequence']), (arr.shape, len(item['sequence']))
        assert np.all((arr >= 0) & (arr <= 1))
        np.save(out/'plddt_0_100.npy', arr.reshape(-1)*100)
        # Independently verify written chain sequence and residue coverage.
        from Bio.PDB import MMCIFParser
        from Bio.SeqUtils import seq1
        cif = out/'prediction.cif' if model_name == 'esmfold2' else result.cif_paths[0]
        structure = MMCIFParser(QUIET=True).get_structure('prediction', str(cif))
        residues = [r for r in structure.get_residues() if r.id[0] == ' ']
        observed = ''.join(seq1(r.resname) for r in residues)
        assert observed == item['sequence'], (item['sequence_id'], observed)

        record = dict(id=item['sequence_id'], model=model_name, seed=seed, sequence=item['sequence'], shape=list(arr.shape), mean=float(arr.mean()), minimum=float(arr.min()), maximum=float(arr.max()), seconds=time.time()-start, status='sequence_and_plddt_validated')
        (out/'complete.json').write_text(json.dumps(record, indent=2))
        print(json.dumps(record), flush=True)
