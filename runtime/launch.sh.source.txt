#!/bin/bash
set -u
source <PROTEIN_TOOLS_ROOT>/model-env.sh
cd <PROTEIN_TOOLS_ROOT>/stability-run-20260916
export OMP_NUM_THREADS=8 MKL_NUM_THREADS=8 CUDA_VISIBLE_DEVICES=0
mkdir -p logs
exec 9>run.lock
flock -n 9 || exit 1
date -Is > started.txt
nvidia-smi --query-gpu=timestamp,memory.used,utilization.gpu,utilization.memory --format=csv -l 10 > logs/gpu.csv &
monitor=$!
trap 'kill "$monitor" 2>/dev/null || true' EXIT
for model in esmfold2 chai1 boltz2; do
  "$PROTEIN_TOOLS/envs/$model/bin/python" -m pip freeze > "logs/$model-versions.txt" 2>&1
done
(
  for seed in 42 43 44; do
    "$PROTEIN_TOOLS/envs/boltz2/bin/boltz" predict boltz_inputs --out_dir "boltz2/$seed" --cache "$PROTEIN_TOOLS/weights/boltz2" --model boltz2 --accelerator gpu --devices 1 --recycling_steps 3 --sampling_steps 200 --diffusion_samples 1 --max_parallel_samples 1 --num_workers 4 --seed "$seed" --write_full_pae --no_kernels > "logs/boltz2-$seed.log" 2>&1
    code=$?
    echo "boltz2 seed=$seed exit=$code $(date -Is)" >> status.log
    [ "$code" -eq 0 ] || exit "$code"
  done
) &
boltz_pid=$!
"$PROTEIN_TOOLS/envs/chai1/bin/python" -u run_model.py chai1 > logs/chai1.log 2>&1 &
chai_pid=$!
echo "boltz_pid=$boltz_pid chai_pid=$chai_pid" > workers.txt
wait "$chai_pid"; echo "chai1 exit=$? $(date -Is)" >> status.log
wait "$boltz_pid"; echo "boltz2 queue exit=$? $(date -Is)" >> status.log
# ESMC-6B uses substantially more memory; give ESMFold2 the whole GPU.
export OMP_NUM_THREADS=16 MKL_NUM_THREADS=16
"$PROTEIN_TOOLS/envs/esmfold2/bin/python" -u run_model.py esmfold2 > logs/esmfold2.log 2>&1
echo "esmfold2 exit=$? $(date -Is)" >> status.log
date -Is > queue_finished.txt
